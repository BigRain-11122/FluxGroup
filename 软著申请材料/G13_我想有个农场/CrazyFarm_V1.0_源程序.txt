        /// <summary>English lookup (missing key falls back to the key itself
        /// - machine-readable, never a crash).</summary>
        public static string EnText(string key)
            => En.TryGetValue(key, out var v) && !string.IsNullOrEmpty(v) ? v : key;
        /// <summary>Parses the zh-CN data file payload into a key-to-text
        /// map (values must be non-empty strings; anything else drops the
        /// entry so the lookup degrades to English, never to garbage).</summary>
        public static Dictionary<string, string> ParseZh(string json)
        {
            var map = new Dictionary<string, string>();
            if (string.IsNullOrEmpty(json)) return map;
            try
            {
                if (!(MiniJson.Deserialize(json) is Dictionary<string, object> d)) return map;
                foreach (var kv in d)
                    if (!string.IsNullOrEmpty(kv.Key) && kv.Value is string s && !string.IsNullOrEmpty(s))
                        map[kv.Key] = s;
            }
            catch
            {
                // corrupt data file = empty map = English fallback face
                return new Dictionary<string, string>();
            }
            return map;
        }
        /// <summary>LocaleService-compatible lookup delegate over a parsed
        /// zh map: zh-CN hits read the data file, everything else (and any
        /// miss) reads the English source. Wire-ready for the T005 UI face.</summary>
        public static Func<string, string, string> LookupWithZh(Dictionary<string, string> zh)
        {
            return (lang, key) =>
            {
                if (lang == "zh-CN" && zh != null && zh.TryGetValue(key, out var v) && !string.IsNullOrEmpty(v))
                    return v;
                return EnText(key);
            };
        }

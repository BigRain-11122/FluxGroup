    /// <summary>
    /// S00_Boot场景宿主：注册Mock服务→跑七步→切主场景（T001④）。
    /// 失败则停留Boot场景并打错误日志（重试页UI属T001b后；Boot校时失败策略=重试页禁卡死）。
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }
}

        /// <summary>三合一：A+B（自动找第三块同品类同档）→ A 位升一档。</summary>
        public bool TryMerge(int slotA, int slotB)
        {
            if (slotA < 0 || slotA >= Slots.Count) return false;
            if (slotB < 0 || slotB >= Slots.Count || slotA == slotB) return false;
            var a = Slots[slotA];
            var b = Slots[slotB];
            if (a.Category != b.Category || a.Tier != b.Tier) return false;
            if (a.Tier >= SalvageConfig.TierTreasure) return false;   // 珍品不可再合
            int third = -1;
            for (int i = 0; i < Slots.Count; i++)
            {
                if (i == slotA || i == slotB) continue;
                if (Slots[i].Category == a.Category && Slots[i].Tier == a.Tier) { third = i; break; }
            }
            if (third < 0) return false;
            // 三块消费：删最大索引先（防位移）
            int[] remove = { slotA, slotB, third };
            Array.Sort(remove);
            for (int i = remove.Length - 1; i >= 0; i--) Slots.RemoveAt(remove[i]);
            var product = new SlotItem { Category = a.Category, Tier = a.Tier + 1 };
            Slots.Insert(Math.Min(remove[0], Slots.Count), product);
            Save();
            ItemMerged?.Invoke(Math.Min(remove[0], Slots.Count - 1), product);
            return true;
        }
        /// <summary>取出台上物品（上架入馆）。</summary>
        public SlotItem TakeAt(int slot)
        {
            if (slot < 0 || slot >= Slots.Count) return null;
            var item = Slots[slot];
            Slots.RemoveAt(slot);
            Save();
            return item;
        }

    /// <summary>
    /// 捞个宝离线结算（P03 OfflinePay 同族范式·8h 帽）：离线时船照常航行（时间戳航程天然兼容）、
    /// 门票照常收入；领取一次性/会话、走真实账本（MuseumHall 入账）、戳只在领取时前移。
    /// 红线12：PlayerPrefs=权威。

    public static class SalvageOffline
    {
        public const string SeenKey = "p07_offline_seen_ms";
        public const int MinClaimMinutes = 60;
        static readonly long MsPerMinute = 60_000L;
        public static void Stamp(long nowMs)
        {
            UnityEngine.PlayerPrefs.SetString(SeenKey, nowMs.ToString());
            UnityEngine.PlayerPrefs.Save();
        }
        public static long LastSeen()
        {
            var raw = UnityEngine.PlayerPrefs.GetString(SeenKey, "");
            return long.TryParse(raw, out long v) ? v : 0;
        }
        public static void ClearForTests()
            => UnityEngine.PlayerPrefs.DeleteKey(SeenKey);
        public sealed class Claim

            public long ElapsedMs { get; }
            public bool Claimable => ElapsedMs >= MinClaimMinutes * MsPerMinute;
            public bool Consumed { get; private set; }
            Claim(long elapsedMs) { ElapsedMs = elapsedMs; }
            public static Claim Open(long nowMs)
            {
                long seen = LastSeen();
                if (seen <= 0) return new Claim(0);
                return new Claim(nowMs - seen);
            }
            /// <summary>领取：门票收入按分钟结算（8h 帽）+全部到点船自动回收入台。
            /// 返回结算分钟数，-1=无可领/已领。</summary>
            public int TryCollect(SalvageFleet fleet, RestoreMini bench, MuseumHall hall, bool highTide)
            {
                if (Consumed || !Claimable) return -1;
                long minutes = ElapsedMs / MsPerMinute;
                const long Cap = SalvageConfig.OfflineCapHours * 60L;
                if (minutes > Cap) minutes = Cap;
                hall.Tick(minutes);   // 门票离线收入（走真实账本）
                // 到点船自动回收→残件入修复台/成品入馆藏
                for (int fid = 1; fid <= SalvageConfig.FleetCount; fid++)
                {
                    var loot = fleet.TryCollect(fid, long.MaxValue, highTide);
                    foreach (var l in loot)
                    {
                        for (int i = 0; i < l.Count; i++)
                        {
                            if (l.Tier >= SalvageConfig.TierRestored) hall.Place(l.Category, l.Tier);
                            else bench.Deposit(l.Category, l.Tier);
                        }
                    }
                }
                Stamp(LastSeen() + minutes * MsPerMinute);
                Consumed = true;
                return (int)minutes;
            }
        }
    }

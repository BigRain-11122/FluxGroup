                             it.tier == TierAbsurd || it.tier == TierHidden;
                if (!tierOk) problems.Add(tag + " tier must be common/rare/absurd/hidden, got " + it.tier);
                if (string.IsNullOrEmpty(it.stem)) problems.Add(tag + " missing stem");
                else if (!stems.Add(it.stem)) problems.Add(tag + " duplicate stem " + it.stem);
                if (string.IsNullOrEmpty(it.nameKey)) problems.Add(tag + " missing nameKey");
            }
            for (int t = 0; t < TierOrder.Length; t++)
            {
                int n = CountTier(TierOrder[t]);
                if (n != ItemsPerTier)
                {
                    problems.Add("tier " + TierOrder[t] + " must hold exactly " + ItemsPerTier +
                                 " items (numeric pack sec1 pool), got " + n);

        static double HalfEnergySec(float[] buf)

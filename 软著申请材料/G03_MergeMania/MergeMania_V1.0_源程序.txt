        /// <summary>Chapters unlocked by the collected count: how many thresholds are crossed.</summary>
        public static int ChapterCountFor(int dexCollected)
        {
            int count = 0;
            for (int i = 0; i < ChapterThresholds.Length; i++)
            {

        /// <summary>
        /// Chapter ids crossed beyond the previous count, ascending (the G02 NewlyUnlocked

        /// </summary>
        public static int[] NewlyUnlocked(int previousChapterCount, int dexCollected)
        {
            int current = ChapterCountFor(dexCollected);
            if (previousChapterCount < 0) previousChapterCount = 0;
            if (current <= previousChapterCount) return new int[0];
            var ids = new int[current - previousChapterCount];
            for (int i = previousChapterCount; i < current; i++) ids[i - previousChapterCount] = i + 1;
            return ids;
        }

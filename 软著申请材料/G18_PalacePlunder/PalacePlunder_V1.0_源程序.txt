        public int LongestCourtStreak()
        {
            var days = new SortedSet<int>();
            var events = _sim.Events;
            for (int i = 0; i < events.Count; i++)
            {
                var e = events[i];
                if (e.Name == "spin") days.Add(PalaceSim.DayIndexOf(e.Time));
            }
            int best = 0, run = 0, prev = int.MinValue;
            foreach (int d in days)
            {
                run = (d == prev + 1) ? run + 1 : 1;
                if (run > best) best = run;
                prev = d;
            }
            return best;
        }

            if (string.IsNullOrEmpty(detail)) return "";

            if (ix < 0) return "";

        static T RowById<T>(T[] rows, string id, Func<T, string> keyOf) where T : class
        {
            if (string.IsNullOrEmpty(id) || rows == null) return null;

            Assert.AreEqual(ticks, (int)(System.Math.Floor(end / 90.0) - System.Math.Floor(start / 90.0)),
                "tick counting must stay the boundary-floor law (frame-rate independent)");
        }

            Assert.AreEqual(1f, WheelViewHost.EaseOutCubic(1f));

            for (int i = 1; i <= 20; i++)
            {
                float u = i / 20f;
                float v = WheelViewHost.EaseOutCubic(u);
                Assert.GreaterOrEqual(v, prev, "the ease must be monotonic (never backtrack)");
                Assert.LessOrEqual(v, 1f, "the ease must never overshoot the end rotation");

            }

        [Test]
        public void Spin_Tick_Synthesis_Is_Deterministic_And_Caliber_Capped()
        {
            var a = G18Sfx.Synth(G18Sfx.SpinTick);
            var b = G18Sfx.Synth(G18Sfx.SpinTick);
            Assert.AreEqual(a.Length, b.Length, "the tick render must be deterministic");
            Assert.Greater(a.Length, 0, "the tick must render actual samples");
            float peak = 0f;
            for (int i = 0; i < a.Length; i++)
            {
                Assert.AreEqual(a[i], b[i], "sample " + i + " must be bit-identical across renders");
                float abs = a[i] < 0f ? -a[i] : a[i];
                if (abs > peak) peak = abs;
            }
            Assert.Greater(peak, 0.05f, "the tick must be audible, not silence");
            Assert.LessOrEqual((double)peak, G18Sfx.SamplePeakCap,
                "sample peak must keep the -3.6 dBFS caliber cap (true-peak margin law)");
        }

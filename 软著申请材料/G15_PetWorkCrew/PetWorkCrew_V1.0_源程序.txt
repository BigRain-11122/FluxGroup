    /// <summary>
    /// S00_Boot scene host: register mock services -> run seven steps -> switch to main scene.
    /// On failure it stays on S00_Boot and logs the error (no dead-end while loops).
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }
}

    /// Seven-step boot sequence: Logger -> Time -> Config -> Save -> Locale -> Ads.Preload -> SwitchMain,
    /// each step logs "[BOOT] n/7 name ok|FAIL". A step exception never blocks later steps.
    /// </summary>
    public sealed class BootSequence
    {
        public const int TotalSteps = 7;
        readonly Func<bool> _switchScene;
        readonly List<string> _lines = new();
        public BootSequence(Func<bool> switchScene)
        {
            _switchScene = switchScene;
        }
        public IReadOnlyList<string> Lines => _lines;
        public bool RunAll()
        {
            bool ok = true;
            ok &= Run(1, "Logger", LoggerStep);
            ok &= Run(2, "Time", TimeStep);
            ok &= Run(3, "Config", ConfigStep);
            ok &= Run(4, "Save", SaveStep);
            ok &= Run(5, "ILocale", LocaleStep);
            ok &= Run(6, "Ads.Preload", AdsStep);
            ok &= Run(7, "SwitchMain", () => _switchScene != null && _switchScene());
            return ok;
        }
        bool Run(int n, string name, Func<bool> step)
        {
            bool result;
            try
            {
                result = step();
            }
            catch (Exception e)
            {
                Logger.Error("BOOT", $"step {n} {name} exception: {e.Message}");
                result = false;
            }
            string line = $"[BOOT] {n}/{TotalSteps} {name} {(result ? "ok" : "FAIL")}";
            _lines.Add(line);
            Logger.Info("BOOT", line);
            return result;
        }

        /// <summary>Placeholder count of a template (defensive: -1 on malformed braces).</summary>
        public static int PlaceholderCount(string template)
        {
            if (string.IsNullOrEmpty(template)) return 0;
            int max = -1;
            for (int i = 0; i < template.Length; i++)
            {
                if (template[i] != '{') continue;
                int close = template.IndexOf('}', i + 1);
                if (close < 0) return -1;
                int n;
                if (!int.TryParse(template.Substring(i + 1, close - i - 1), out n)) return -1;
                if (n > max) max = n;
                i = close;
            }

        }

            if (!TryFromB36(check, out want) || want != (long)Fnv1a(payload)) return false;
            string[] f = payload.Split('.');
            if (f.Length != 8 || f[0] != ShareCodePrefix) return false;
            long seq, pet1, job1, workSec, coinCents, starCents;
            if (!TryFromB36(f[2], out seq) || !TryFromB36(f[3], out pet1) || !TryFromB36(f[4], out job1)
                || !TryFromB36(f[5], out workSec) || !TryFromB36(f[6], out coinCents) || !TryFromB36(f[7], out starCents))
                return false;
            if ((f[1] != "W" && f[1] != "A") || pet1 < 0 || pet1 > PetData.PetCount
                || job1 < 0 || job1 > PetData.JobCount || seq < 1)
                return false;
            rec = new PostcardRecord
            {
                Seq = (int)seq,
                Kind = f[1] == "W" ? PetData.PostcardKind.Wage : PetData.PostcardKind.Away,
                Pet = (int)(pet1 - 1),
                Job = (int)(job1 - 1),
                WorkSec = workSec,
                Coin = coinCents / 100.0,
                StarDust = starCents / 100.0,
            };
            return true;
        }
    }

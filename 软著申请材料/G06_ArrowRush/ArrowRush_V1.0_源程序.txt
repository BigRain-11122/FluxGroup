    /// Telemetry assembly point for G06 (task S2-T002, G03TelemetryWire paradigm):
    /// DebugLogSink is always attached; the Wx analytics sink joins when a reporter

    {
        public static TelemetryService CreateService(IAnalyticsReporter reporter = null)
        {
            var telemetry = new TelemetryService();
            telemetry.AddSink(new DebugLogSink());
            if (reporter == null && ServiceLocator.TryGet<IAnalyticsReporter>(out var fromLocator))
                reporter = fromLocator;
            if (reporter != null) telemetry.AddSink(new WxAnalyticsSink(reporter));
            return telemetry;
        }
    }

        static Dictionary<string, string> ParseFlatJson(string json)
        {
            var result = new Dictionary<string, string>(StringComparer.Ordinal);
            if (string.IsNullOrEmpty(json)) return result;
            int i = 0;
            int n = json.Length;
            while (i < n)
            {
                int keyStart = json.IndexOf('"', i);
                if (keyStart < 0) break;
                int keyEnd = json.IndexOf('"', keyStart + 1);
                if (keyEnd < 0) break;
                int colon = json.IndexOf(':', keyEnd + 1);
                if (colon < 0) break;
                int valStart = json.IndexOf('"', colon + 1);
                if (valStart < 0) break;
                int valEnd = valStart + 1;
                var val = new StringBuilder();
                while (valEnd < n)
                {
                    char c = json[valEnd];
                    if (c == '\\')
                    {
                        char next = valEnd + 1 < n ? json[valEnd + 1] : '\0';
                        if (next == 'n') val.Append('\n');
                        else if (next == 't') val.Append('\t');
                        else val.Append(next);
                        valEnd += 2;
                        continue;
                    }
                    if (c == '"') break;
                    val.Append(c);
                    valEnd++;
                }
                result[json.Substring(keyStart + 1, keyEnd - keyStart - 1)] = val.ToString();
                i = valEnd + 1;
            }

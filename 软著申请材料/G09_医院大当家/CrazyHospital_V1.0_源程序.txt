    public static class SceneIds
    {
        public const string Boot = "S00_Boot";
        public const string Main = "S09_Main";

    /// <summary>
    /// S00_Boot scene host: register mock services -> run seven steps -> switch to main scene.
    /// On failure it stays on S00_Boot and logs the error (no dead-end while loops).
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }
}

    /// Seven-step boot sequence: Logger -> Time -> Config -> Save -> Locale -> Ads.Preload -> SwitchMain,
    /// each step logs "[BOOT] n/7 name ok|FAIL". A step exception never blocks later steps.
    /// </summary>
    public sealed class BootSequence
    {
        public const int TotalSteps = 7;
        readonly Func<bool> _switchScene;
        readonly List<string> _lines = new();
        public BootSequence(Func<bool> switchScene)
        {
            _switchScene = switchScene;
        }
        public IReadOnlyList<string> Lines => _lines;
        public bool RunAll()
        {
            bool ok = true;
            ok &= Run(1, "Logger", LoggerStep);
            ok &= Run(2, "Time", TimeStep);
            ok &= Run(3, "Config", ConfigStep);
            ok &= Run(4, "Save", SaveStep);
            ok &= Run(5, "ILocale", LocaleStep);
            ok &= Run(6, "Ads.Preload", AdsStep);
            ok &= Run(7, "SwitchMain", () => _switchScene != null && _switchScene());
            return ok;
        }
        bool Run(int n, string name, Func<bool> step)
        {
            bool result;
            try
            {
                result = step();
            }
            catch (Exception e)
            {
                Logger.Error("BOOT", $"step {n} {name} exception: {e.Message}");
                result = false;
            }
            string line = $"[BOOT] {n}/{TotalSteps} {name} {(result ? "ok" : "FAIL")}";
            _lines.Add(line);
            Logger.Info("BOOT", line);
            return result;
        }

        public void Advance(double dt)
        {
            if (dt <= 0) return;
            if (TimeOf("evt_hospital_hook") < 0)
                Stamp("evt_hospital_hook");   // the hook owns t0 (facade + street queue + caption)
            _admitBlocked = false;   // the registration retry law is per-call (a refused admit waits for the next Advance)
            double remaining = dt;
            while (remaining > 0)
            {
                double seg = remaining;
                double next = NextBeatTime();
                if (next - _t < seg) seg = next - _t;   // slice at the beat, never past it
                if (seg > 0)
                {

        /// <summary>English lookup (missing key falls back to the key itself
        /// - machine-readable, never a crash).</summary>
        public static string EnText(string key)
            => En.TryGetValue(key, out var v) && !string.IsNullOrEmpty(v) ? v : key;
        /// <summary>Parses the zh-CN data file payload into a key-to-text
        /// map (values must be non-empty strings; anything else drops the
        /// entry so the lookup degrades to English, never to garbage).</summary>
        public static Dictionary<string, string> ParseZh(string json)
        {
            var map = new Dictionary<string, string>();
            if (string.IsNullOrEmpty(json)) return map;
            try
            {
                if (!(MiniJson.Deserialize(json) is Dictionary<string, object> d)) return map;
                foreach (var kv in d)
                    if (!string.IsNullOrEmpty(kv.Key) && kv.Value is string s && !string.IsNullOrEmpty(s))
                        map[kv.Key] = s;
            }
            catch
            {
                // corrupt data file = empty map = English fallback face
                return new Dictionary<string, string>();
            }
            return map;
        }
        /// <summary>LocaleService-compatible lookup delegate over a parsed
        /// zh map: zh-CN hits read the data file, everything else (and any
        /// miss) reads the English source. Wire-ready for the T005 UI face.</summary>
        public static Func<string, string, string> LookupWithZh(Dictionary<string, string> zh)
        {
            return (lang, key) =>
            {
                if (lang == "zh-CN" && zh != null && zh.TryGetValue(key, out var v) && !string.IsNullOrEmpty(v))
                    return v;
                return EnText(key);
            };
        }

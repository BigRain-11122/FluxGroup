    /// S00_Boot scene host: register mock services -> run seven steps -> switch to main scene.
    /// On failure it stays on S00_Boot and logs the error (no dead-end while loops).
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }

using UnityEngine;
namespace Sy.G02

    /// <summary>
    /// Repair economy math (GDD section 5 + prototype parity):
    /// reward pct = 100 - floor(mistakes / penaltyPerMistakes) * penaltyPct, floored at minRewardPct;
    /// fee = era.repairFee * (faults >= 2 ? doubleFaultFeeMult : 1) * rewardPct / 100, rounded;
    /// stars = perfectStars when zero mistakes else normalStars.
    /// </summary>
    public static class Economy
    {
        public static int RewardPct(int mistakes, GameCatalog catalog)
        {
            int q = 100 - (mistakes / catalog.Economy.penaltyPerMistakes) * catalog.Economy.penaltyPct;
            return Mathf.Max(catalog.Economy.minRewardPct, q);
        }
        public static int FeeFor(int eraId, int faultCount, int mistakes, GameCatalog catalog)
        {
            var era = catalog.Era(eraId);
            if (era == null) return 0;
            float fee = era.repairFee * (faultCount >= 2 ? catalog.Economy.doubleFaultFeeMult : 1f);
            return Mathf.RoundToInt(fee * RewardPct(mistakes, catalog) / 100f);
        }
        public static int StarsFor(int mistakes, GameCatalog catalog)
        {
            return mistakes == 0 ? catalog.Economy.perfectStars : catalog.Economy.normalStars;
        }
    }

    /// Era ids are 0-based (era 0 = E1, always unlocked).
    /// </summary>
    public static class G02ChapterGate
    {
        /// <summary>
        /// Era ids that the star total unlocks beyond the previous count, ascending.
        /// Era 1 is always unlocked, so the previous count is clamped to at least 1.
        /// </summary>
        public static int[] NewlyUnlocked(int previousEraCount, int starsTotal, GameCatalog catalog)
        {
            if (catalog == null) return new int[0];
            int previous = previousEraCount < 1 ? 1 : previousEraCount;
            int current = catalog.UnlockedEraCount(starsTotal < 0 ? 0 : starsTotal);
            if (current <= previous) return new int[0];
            var ids = new int[current - previous];
            for (int i = previous; i < current; i++) ids[i - previous] = i;
            return ids;
        }
        /// <summary>
        /// Apply the unlock check and return the era count to persist. Publishes one
        /// EraUnlockedEvent per newly unlocked era in ascending order; a null bus is
        /// tolerated (count-only pass, no throw) so callers without a bus stay safe.
        /// </summary>
        public static int ApplyUnlocks(SyEventBus bus, int previousEraCount, int starsTotal, GameCatalog catalog)
        {
            if (catalog == null) return previousEraCount < 1 ? 1 : previousEraCount;
            int previous = previousEraCount < 1 ? 1 : previousEraCount;
            int current = catalog.UnlockedEraCount(starsTotal < 0 ? 0 : starsTotal);
            if (bus != null)
            {
                for (int i = previous; i < current; i++)
                {
                    var era = catalog.Era(i);
                    bus.Publish(new EraUnlockedEvent
                    {
                        eraId = i,
                        years = era != null ? era.years : "",
                        unlockStars = era != null ? era.unlockStars : 0,
                        starsTotal = starsTotal < 0 ? 0 : starsTotal,
                    });
                }
            }
            return current > previous ? current : previous;
        }
    }

                    guard++;

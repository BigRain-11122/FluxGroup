        {

            ok &= Run(2, "Time", TimeStep);
            ok &= Run(3, "Config", ConfigStep);
            ok &= Run(4, "Save", SaveStep);
            ok &= Run(5, "ILocale", LocaleStep);
            ok &= Run(6, "Ads.Preload", AdsStep);

            _lines.Add(line);

        public static AudioClip EnsureClip()
        {
            if (click != null) return click;
            const float dur = 0.09f;
            int count = (int)(SampleRate * dur);
            var data = new float[count];
            for (int i = 0; i < count; i++)
            {
                float t = (float)i / SampleRate;
                data[i] = 0.9f * Tick(t, 0.000f, 900f) + 0.7f * Tick(t, 0.035f, 600f);
            }
            click = AudioClip.Create(ClipName, count, 1, SampleRate, false);
            click.SetData(data, 0);
            return click;
        }
        static float Tick(float t, float start, float freq)
        {
            float dt = t - start;
            if (dt < 0f) return 0f;
            return Mathf.Exp(-dt * 90f) * Mathf.Sin(2f * Mathf.PI * freq * dt);
        }
        public static void EnsureSource(Component host)
        {
            if (source != null) return;
            source = host.gameObject.AddComponent<AudioSource>();
            source.playOnAwake = false;
        }

        public static void PlayClick()

            if (source == null || !Application.isPlaying) return;
            if (click == null) EnsureClip();

        }

using UnityEngine;

    public sealed class PopFx : MonoBehaviour
    {
        public const float Life = 0.5f;
        public float delay;          // cascade stagger: carried screws tumble one after another
        public float vx = 2.4f;
        public float vy = 3.4f;
        public float spinDegPerSec = 540f;
        float age;
        SpriteRenderer sr;
        Color color0;
        void Awake()
        {
            sr = GetComponent<SpriteRenderer>();
            if (sr != null) color0 = sr.color;
        }
        void Update()
        {
            if (delay > 0f)
            {
                delay -= Time.deltaTime;
                if (delay > 0f) return;
            }
            age += Time.deltaTime;
            float k = age / Life;
            transform.localPosition += new Vector3(vx, vy, 0f) * Time.deltaTime;
            vy -= 9.8f * Time.deltaTime;   // gravity: a short arc, then it drops off the board
            transform.Rotate(0f, 0f, spinDegPerSec * Time.deltaTime);
            if (sr != null)
            {
                Color c = color0;
                c.a = Mathf.Max(0f, 1f - k);
                sr.color = c;
            }
            if (age >= Life) Destroy(gameObject);
        }
    }

using System;
using System.Collections.Generic;
namespace GimmeAll.Core
{
    /// <summary>
    /// Typed in-process event bus. Main-thread only.
    /// Modules must communicate across assembly boundaries ONLY through this bus,
    /// the registries, or the service interfaces in Core. Never reference another module's code.
    /// </summary>
    public static class GameEventBus
    {
        private static readonly Dictionary<Type, List<Delegate>> Handlers = new Dictionary<Type, List<Delegate>>(16);
        public static void Subscribe<T>(Action<T> handler)
        {
            if (handler == null) return;
            if (!Handlers.TryGetValue(typeof(T), out var list))
            {
                list = new List<Delegate>(4);
                Handlers.Add(typeof(T), list);
            }
            if (!list.Contains(handler)) list.Add(handler);
        }
        public static void Unsubscribe<T>(Action<T> handler)
        {
            if (handler == null) return;
            if (Handlers.TryGetValue(typeof(T), out var list))
            {
                list.Remove(handler);
                if (list.Count == 0) Handlers.Remove(typeof(T));
            }
        }
        public static void Publish<T>(in T evt)
        {
            if (!Handlers.TryGetValue(typeof(T), out var list) || list.Count == 0) return;
            // Iterate defensively: handlers may unsubscribe during publish.
            for (int i = list.Count - 1; i >= 0; i--)
            {
                if (i >= list.Count) continue;
                if (list[i] is Action<T> action) action(evt);
            }
        }
        public static void ClearAll()
        {
            Handlers.Clear();
        }
    }
}

        public static void Kick(RectTransform target)

            var nudge = target.GetComponent<UiNudge>();
            if (nudge == null) nudge = target.gameObject.AddComponent<UiNudge>();

            nudge._t = 0f;

        private void Update()
        {

            _t += Time.unscaledDeltaTime;
            if (_t >= Duration) { _rt.anchoredPosition = _base; Destroy(this); return; }
            float phase = _t / Duration;
            float x = Mathf.Sin(phase * Mathf.PI * Beats * 2f) * Amplitude * (1f - phase);
            _rt.anchoredPosition = _base + new Vector2(x, 0f);
        }

        /// <summary>程序化 128² 软边圆涟漪（哑白，色由 Image.color 染——CollectionScreen 剪影同款）。</summary>
        private static Sprite BuildRippleSprite()
        {
            const int Size = 128;
            const float Radius = 56f;
            const float Edge = 6f;
            var tex = new Texture2D(Size, Size, TextureFormat.RGBA32, false);
            tex.wrapMode = TextureWrapMode.Clamp;
            tex.filterMode = FilterMode.Bilinear;
            var pixels = new Color32[Size * Size];
            int k = 0;
            for (int y = 0; y < Size; y++)
            {
                for (int x = 0; x < Size; x++)
                {
                    float dx = x + 0.5f - Size * 0.5f;
                    float dy = y + 0.5f - Size * 0.5f;
                    float dist = Mathf.Sqrt(dx * dx + dy * dy);
                    float alpha = Mathf.Clamp01((Radius - dist) / Edge);
                    pixels[k++] = new Color32(255, 255, 255, (byte)(alpha * 255f + 0.5f));
                }
            }
            tex.SetPixels32(pixels);
            tex.Apply(false, false);
            var sprite = Sprite.Create(tex, new Rect(0f, 0f, Size, Size),
                new Vector2(Size * 0.5f, Size * 0.5f), 100f);
            sprite.name = "tut_ripple_circle";
            return sprite;
        }

        public static float EaseOut01(float k)
        {
            if (k < 0f) k = 0f;
            if (k > 1f) k = 1f;
            float inv = 1f - k;
            return 1f - inv * inv;
        }
        /// <summary>EaseOutCubic：数字滚动用（0.3s 内减速抵达·CEO 手册「数字不瞬跳」正身）。</summary>
        public static float EaseOutCubic01(float k)
        {
            if (k < 0f) k = 0f;
            if (k > 1f) k = 1f;
            float inv = 1f - k;
            return 1f - inv * inv * inv;

        public static float EaseIn01(float k)
        {
            if (k < 0f) k = 0f;
            if (k > 1f) k = 1f;
            return k * k;
        }
        /// <summary>Pop（EaseOutBack 族·温和过冲）：0→1 且中段冲过 1 到峰约 1.10 后回落恰好 1
        /// ——弹窗 0.8→1 与屏入 0.95→1 的「Q 弹」曲线（与 JuiceCurves.SpawnPop01 同包络口不同实现）。</summary>
        public static float Pop01(float k)
        {
            if (k < 0f) k = 0f;
            if (k > 1f) k = 1f;
            // easeOutBack 经典参数 s=1.70158：峰≈1.10，端点 0/1 精确
            const float s = 1.70158f;
            float t = k - 1f;
            return 1f + (s + 1f) * t * t * t + s * t * t;
        }
        /// <summary>白闪包络（U305 转场律）：0→急升到峰（前 20%）→平台保持 →缓落回 0。
        /// hold=满亮段时长；fall=落败尾长。全部 unscaled 语义、确定可测。</summary>
        public static float FlashAlphaAt(float t, float rise, float hold, float fall, float peak)
        {
            if (t < 0f) return 0f;
            if (t < rise) return peak * EaseOut01(t / rise);
            if (t < rise + hold) return peak;
            float k = (t - rise - hold) / fall;
            if (k >= 1f) return 0f;
            return peak * (1f - EaseIn01(k));
        }

        private void Open()
        {
            Refresh();

            UiFade.Ensure(_content.gameObject, _panel).PlayIn();   // W21-UX 模态过渡
        }
        private void Close()
        {
            UiFade.Ensure(_content.gameObject, _panel).PlayOut();
        }
        private void Refresh()
        {
            int equipped = PlayerPrefs.GetInt(EquipKey, 0);
            for (int i = 0; i < 19; i++)
            {
                bool unlocked = IsUnlocked(i);
                var img = _slotBtn[i].GetComponent<Image>();
                img.color = unlocked ? (i == equipped ? UIRoot.Coral : UIRoot.CoralSoft) : UIRoot.Lilac;
                _slotState[i].text = !unlocked ? "通关第 " + i + " 章解锁"
                    : (i == equipped ? "在身上～" : "");
                _slotState[i].raycastTarget = false;
            }
        }
        private void OnSlotClicked(int idx)
        {
            if (!IsUnlocked(idx)) return;   // 锁定态点不动（零挫败：不弹窗不惩罚）
            PlayerPrefs.SetInt(EquipKey, idx);
            PlayerPrefs.Save();
            PublishEquip(idx);
            Refresh();
        }
        private static void PublishEquip(int chapter)
        {
            if (chapter <= 0)
            {
                GameEventBus.Publish(new OutfitEquipped { Chapter = 0, R = -1f, G = -1f, B = -1f });
                return;
            }
            Color c = ChapterPalette.ButtonOf(chapter);   // 章色 tint 由 UI 算好随事件走（Gameplay 零依赖）
            GameEventBus.Publish(new OutfitEquipped { Chapter = chapter, R = c.r, G = c.g, B = c.b });
        }

    /// <summary>
    /// S00_Boot场景宿主：注册Mock服务→跑七步→切S02_Main（任务书T001④）。
    /// 失败则停留Boot场景并打错误日志（重试页UI属T001b后；Boot校时失败策略=重试页禁卡死）。
    /// U185（19号§二F组）：启动页合规面——BootServices就绪后即建LaunchComplianceFace
    /// （健康游戏忠告全文+批准信息块，全部文案走cfg_i18n）；步⑦仅armed，实际切场景由
    /// 「最短展示窗(3s)或点击进入」门驱动——真实启动必见忠告，流程门禁为sim级不经本场景零漂移。
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        bool _switchArmed;
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var face = LaunchComplianceFace.Build();
            float shownAt = Time.realtimeSinceStartup;
            var sequence = new BootSequence(() => { _switchArmed = true; return true; });
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            while (!(_switchArmed && (face.EnterTapped
                || Time.realtimeSinceStartup - shownAt >= LaunchComplianceFace.MinDisplaySeconds)))
                yield return null;
            yield return LoadMainRoutine();
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }
}

        {
            bool ok = true;
            ok &= Run(1, "Logger", LoggerStep);
            ok &= Run(2, "Time", TimeStep);
            ok &= Run(3, "Config", ConfigStep);

        }
        bool Run(int n, string name, Func<bool> step)
        {
            bool result;
            try
            {
                result = step();
            }
            catch (Exception e)

                result = false;
            }
            string line = $"[BOOT] {n}/{TotalSteps} {name} {(result ? "ok" : "FAIL")}";
            _lines.Add(line);
            Logger.Info("BOOT", line);
            return result;

    /// <summary>
    /// U184 font-chain fix (audit B1/B2): resolve the UI font through the
    /// locale FontSet instead of hardcoding the engine built-in font.
    /// Why: LegacyRuntime.ttf carries no CJK glyphs and WebGL/WeChat builds
    /// have no OS font fallback, so every Chinese label would render blank
    /// on device (editor stays green via OS fallback - a false-comfort
    /// mask). The bundled subset asset P02FontCJK.ttf (Noto Sans SC OFL,
    /// subset to this game's own character census, 168KB) covers the whole
    /// UI vocabulary; X027 law: missing asset falls back to the built-in
    /// font + one warning, never breaks a build or gate.
    /// </summary>
    public static class UiFont
    {
        const string Fallback = "LegacyRuntime.ttf";
        static bool _warned;
        public static Font Resolve()
        {
            string name = null;
            try
            {
                if (ServiceLocator.TryGet<ILocaleService>(out var locale))
                    name = locale.GetFontSet().NormalFont;
            }
            catch (Exception) { name = null; }
            if (!string.IsNullOrEmpty(name) && name != Fallback)
            {
                var font = Resources.Load<Font>("Fonts/" + name);
                if (font != null) return font;
                if (!_warned)
                {
                    Sy.Core.Logger.Warn("UIFONT", "font asset missing, fallback LegacyRuntime: " + name);
                    _warned = true;
                }
            }
            return Resources.GetBuiltinResource<Font>(Fallback);
        }

            public void Report(string evtId, string fieldsJson)
                => _sink.Add(new KeyValuePair<string, string>(evtId, fieldsJson));
        }
        [Test]
        public async Task TcbBatchSinkFlushesInChunks()
        {
            var sink = new TcbBatchSink(50);
            for (int i = 0; i < 120; i++)
                sink.Write(EventIds.NewbieStep, new Dictionary<string, string> { { "step", "1" } });
            Assert.AreEqual(120, sink.BufferedCount);
            Assert.IsTrue(sink.NeedsFlush);
            var calls = new List<string>();
            var cloud = new CallbackCloud(calls.Add);
            int sent = await sink.FlushAsync(cloud);
            Assert.AreEqual(120, sent);
            Assert.AreEqual(0, sink.BufferedCount);
            Assert.AreEqual(3, calls.Count, "flush batches 50/50/20");
            Assert.AreEqual(0, await sink.FlushAsync(cloud), "empty flush sends nothing");
        }
        sealed class CallbackCloud : ICloudService
        {
            readonly Action<string> _onCall;
            public CallbackCloud(Action<string> onCall) { _onCall = onCall; }
            public Task<string> CallFunction(string functionName, string payloadJson)
            {
                _onCall(functionName);
                return Task.FromResult("{\"ok\":true}");
            }
        }

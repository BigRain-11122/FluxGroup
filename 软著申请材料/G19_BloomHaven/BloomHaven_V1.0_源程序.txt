            bool capped = raw > AccrualCap;

            int credited = Math.Min(Sy.BloomHaven.Garden.GardenSystem.WalletCap - State.Garden.Petals, accrued);

        {

// HudFont.cs
// Pixel-font HUD chain (P2 advisory X1070, FT-011 subset X1301): the
// shipped G19Main face (Fusion Pixel 12px subset, census-covered) is the
// single HUD font seam - every player-facing Text face resolves through
// here. A missing or pruned asset falls back to the builtin face so a
// label never renders dead (X027 null-safe law); the shipped census
// (Design/configs/G19/G19_font_census.txt) covers every i18n value and
// code literal, so the subset can never tofu before I18nCheck group C
// fails the build first.
using UnityEngine;
namespace Sy.BloomHaven.UI
{
    public static class HudFont
    {
        const string ResourcePath = "Fonts/G19Main";
        const string BuiltinFallback = "LegacyRuntime.ttf";
        static Font _cached;
        /// <summary>HUD font face. Cached on first resolve; falls back to
        /// the builtin face when the shipped subset is absent so the UI
        /// never depends on a live asset to render (X027 seam).</summary>
        public static Font Resolve()
        {
            if (_cached != null) return _cached;
            var font = Resources.Load<Font>(ResourcePath);
            if (font == null) font = Resources.GetBuiltinResource<Font>(BuiltinFallback);
            _cached = font;
            return _cached;
        }
    }

// S03 codex, S04 share) build one art node each and apply the produced
// sprite through this seam so the gate can assert the null-safe law in
// one place. Art missing (catalog empty, art pruned, name typo) means the
// node sleeps and the procedural color-block face stays untouched - the
// same contract ArtRuntime.Get already enforces for every consumer.
using UnityEngine;
using UnityEngine.UI;
namespace Sy.BloomHaven.UI
{
    public static class PageArt
    {
        /// <summary>
        /// Build the page-art node (anchored center, behind every sibling)
        /// and apply the sprite. Returns the node whether or not art lives.
        /// </summary>
        public static GameObject BuildNode(Transform parent, string name,
            float x, float y, float w, float h, Sprite art)
        {
            var go = new GameObject(name, typeof(RectTransform), typeof(Image));
            go.transform.SetParent(parent, false);
            var rt = (RectTransform)go.transform;
            rt.sizeDelta = new Vector2(w, h);
            rt.anchoredPosition = new Vector2(x, y);
            rt.SetAsFirstSibling(); // page art sits behind every sibling face
            Apply(go.GetComponent<Image>(), art);
            return go;
        }
        /// <summary>
        /// Null-safe sprite apply. Returns true when the node shows live art;
        /// a null sprite deactivates the node so the procedural look stays.
        /// </summary>
        public static bool Apply(Image target, Sprite art)
        {
            if (target == null) return false;
            if (art == null)
            {
                target.sprite = null;
                target.color = Color.white;
                target.gameObject.SetActive(false);
                return false;
            }
            target.sprite = art;
            target.color = Color.white;
            target.preserveAspect = true;
            target.raycastTarget = false; // backdrop never steals a tap
            target.gameObject.SetActive(true);
            return true;
        }
    }

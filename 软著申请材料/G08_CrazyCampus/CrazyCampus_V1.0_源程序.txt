    /// <summary>
    /// S00_Boot scene host: register mock services -> run seven steps -> switch to main scene.
    /// On failure it stays on S00_Boot and logs the error (no dead-end while loops).
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
            // S1 campus stage mounts itself in code (G08-T005); S2 scene authoring replaces this.
            var host = new GameObject("CampusStageView");
            host.AddComponent<CampusStageView>();
        }
    }

        public static ScheduleWeights Normalized(ScheduleWeights w)
        {
            int a = Clamp(w.study), b = Clamp(w.play), c = Clamp(w.club), d = Clamp(w.rest);
            int sum = a + b + c + d;

            int[] scaled = new int[4];
            for (int i = 0; i < 4; i++) scaled[i] = v[i] * 100 / sum;
            int left = 100 - (scaled[0] + scaled[1] + scaled[2] + scaled[3]);
            while (left > 0)
            {

                int best = 0;
                long bestFrac = -1;
                for (int i = 0; i < 4; i++)
                {

                    if (frac > bestFrac) { bestFrac = frac; best = i; }
                }
                scaled[best]++;
                left--;
            }
            var n = new ScheduleWeights { study = scaled[0], play = scaled[1], club = scaled[2], rest = scaled[3] };

    /// S00_Boot场景宿主：注册Mock服务→跑七步→切主场景（T001④）。
    /// 失败则停留Boot场景并打错误日志（重试页UI属T001b后；Boot校时失败策略=重试页禁卡死）。

    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }

                    _socialButton.gameObject.SetActive(true);
                    break;

                    _socialButton.gameObject.SetActive(true);

        public float CapFor(int warehouseLevel)
            => NightConfig.StorageBase + warehouseLevel * NightConfig.WarehouseCapPerLevel;
        /// <summary>分钟产出→tick 采样（每分钟速率×dt 分钟）。产出即刻入账并帽仓。</summary>
        public void Tick(float dtMinutes, float woodPerMin, float grainPerMin, float ironPerMin, int warehouseLevel)
        {
            if (dtMinutes <= 0f) return;
            float cap = CapFor(warehouseLevel);
            Wood = Mathf.Min(cap, Wood + woodPerMin * ProductionMultiplier * dtMinutes);
            Grain = Mathf.Min(cap, Grain + grainPerMin * ProductionMultiplier * dtMinutes);
            Iron = Mathf.Min(cap, Iron + ironPerMin * ProductionMultiplier * dtMinutes);
            Save();
        }

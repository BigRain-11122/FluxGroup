    /// <summary>
    /// S00_Boot scene host: register mock services -> run seven steps -> switch to main scene.
    /// On failure it stays on S00_Boot and logs the error (no dead-end while loops).
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }

        readonly Func<bool> _switchScene;
        readonly List<string> _lines = new();
        public BootSequence(Func<bool> switchScene)
        {
            _switchScene = switchScene;
        }
        public IReadOnlyList<string> Lines => _lines;
        public bool RunAll()
        {
            bool ok = true;
            ok &= Run(1, "Logger", LoggerStep);
            ok &= Run(2, "Time", TimeStep);
            ok &= Run(3, "Config", ConfigStep);
            ok &= Run(4, "Save", SaveStep);
            ok &= Run(5, "ILocale", LocaleStep);
            ok &= Run(6, "Ads.Preload", AdsStep);
            ok &= Run(7, "SwitchMain", () => _switchScene != null && _switchScene());
            return ok;
        }
        bool Run(int n, string name, Func<bool> step)
        {
            bool result;
            try
            {
                result = step();
            }
            catch (Exception e)
            {
                Logger.Error("BOOT", $"step {n} {name} exception: {e.Message}");
                result = false;
            }
            string line = $"[BOOT] {n}/{TotalSteps} {name} {(result ? "ok" : "FAIL")}";
            _lines.Add(line);
            Logger.Info("BOOT", line);
            return result;
        }

                if (!r.Legal) continue;

            public static bool TryParse(string payload, out SaveFace face)
            {
                face = null;
                if (payload == null) return false;
                string[] sections = payload.Split('|');
                if (sections.Length != 5 || sections[0] != Prefix) return false;
                if (!sections[1].StartsWith("L=", StringComparison.Ordinal) ||
                    !sections[2].StartsWith("S=", StringComparison.Ordinal) ||
                    !sections[3].StartsWith("C=", StringComparison.Ordinal) ||
                    !sections[4].StartsWith("A=", StringComparison.Ordinal)) return false;
                SaveFace f = new SaveFace();
                if (!int.TryParse(sections[1].Substring(2), NumberStyles.None, CultureInfo.InvariantCulture, out f.LevelsCompleted) ||
                    f.LevelsCompleted < 0) return false;
                if (!int.TryParse(sections[2].Substring(2), NumberStyles.None, CultureInfo.InvariantCulture, out f.LevelsSeen) ||
                    f.LevelsSeen < 0) return false;
                if (!int.TryParse(sections[3].Substring(2), NumberStyles.None, CultureInfo.InvariantCulture, out f.ShareCards) ||
                    f.ShareCards < 0) return false;
                if (f.LevelsSeen < f.LevelsCompleted) return false;   // seen implies completed
                string a = sections[4].Substring(2);
                if (a.Length > 0)
                {
                    string[] stamps = a.Split(',');
                    long ts;
                    for (int i = 0; i < stamps.Length; i++)
                    {
                        if (!long.TryParse(stamps[i], NumberStyles.None, CultureInfo.InvariantCulture, out ts) ||
                            ts < 0) return false;
                        f.AdGrantStamps.Add(ts);
                    }
                }
                face = f;
                return true;
            }

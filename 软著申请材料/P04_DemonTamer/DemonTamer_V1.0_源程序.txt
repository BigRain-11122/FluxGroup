    /// <summary>
    /// S00_Boot场景宿主：注册Mock服务→跑七步→切主场景（T001④）。
    /// 失败则停留Boot场景并打错误日志（重试页UI属T001b后；Boot校时失败策略=重试页禁卡死）。

    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }
}

        {
            bool ok = true;
            ok &= Run(1, "Logger", LoggerStep);
            ok &= Run(2, "Time", TimeStep);
            ok &= Run(3, "Config", ConfigStep);
            ok &= Run(4, "Save", SaveStep);

        bool Run(int n, string name, Func<bool> step)
        {
            bool result;
            try
            {
                result = step();

            catch (Exception e)
            {
                Logger.Error("BOOT", $"step {n} {name} exception: {e.Message}");
                result = false;
            }
            string line = $"[BOOT] {n}/{TotalSteps} {name} {(result ? "ok" : "FAIL")}";
            _lines.Add(line);
            Logger.Info("BOOT", line);
            return result;

        public void SyncView(IdleSim sim)
        {
            ReconcileEnemies(sim);
            foreach (var kv in _enemies)
            {
                var v = kv.Value;
                v.Rt.anchoredPosition = new Vector2(FriendX + v.Pos * FieldSpan, RowY[v.Row]);
                v.Fill.fillAmount = v.MaxHp > 0f ? v.Hp / v.MaxHp : 0f;
                if (v.SealReady != v.LastSealReady)
                {
                    v.LastSealReady = v.SealReady;
                    v.SealLabel.gameObject.SetActive(v.SealReady);
                    v.Fill.color = v.SealReady ? PalGold : new Color(0.2f, 0.9f, 0.3f);
                    v.LastAttempts = -1;
                }
                if (v.SealReady && v.LastAttempts != v.Attempts)
                {
                    v.LastAttempts = v.Attempts;
                    int next = v.Attempts + 1;
                    v.SealLabel.text = next >= NewbieConfig.PityGuaranteeCount
                        ? "可封印（第" + next + "/" + NewbieConfig.PityGuaranteeCount + "次·必成）"
                        : "可封印（第" + next + "/" + NewbieConfig.PityGuaranteeCount + "次）";
                }
                if (v.Shielded != v.LastShielded)
                {
                    v.LastShielded = v.Shielded;
                    v.ShieldLabel.gameObject.SetActive(v.Shielded);
                }
            }
            for (int i = 0; i < _friendHpFills.Length; i++)
            {
                var f = sim.Friends[i];
                _friendHpFills[i].fillAmount = f.MaxHp > 0f ? f.Hp / f.MaxHp : 0f;
            }
            bool canSeal = sim.SealReadyEnemy != null;
            if (_sealButton.interactable != canSeal)
            {
                _sealButton.interactable = canSeal;
                _sealButtonImage.color = canSeal ? PalGold : new Color(0.35f, 0.35f, 0.38f);
                _sealLabel.color = canSeal ? PalBlack : new Color(0.7f, 0.7f, 0.7f);
            }
        }

        void ReconcileEnemies(YaotideSim sim)
        {
            _scratch.Clear();
            foreach (var kv in _enemies)
                if (!ContainsEnemy(sim, kv.Key)) _scratch.Add(kv.Key);
            for (int i = 0; i < _scratch.Count; i++)
            {
                Destroy(_enemies[_scratch[i]].Go);
                _enemies.Remove(_scratch[i]);
            }
            for (int i = 0; i < sim.Enemies.Count; i++)
            {
                var e = sim.Enemies[i];
                if (!_enemies.TryGetValue(e.Id, out var v))
                {
                    v = CreateEnemyVisual(e);
                    _enemies[e.Id] = v;
                }
                v.X = e.X;
                v.Y = e.Y;
                v.Hp = e.Hp;
                v.MaxHp = e.MaxHp;
                v.Focused = sim.IsFocused(e);
            }
        }

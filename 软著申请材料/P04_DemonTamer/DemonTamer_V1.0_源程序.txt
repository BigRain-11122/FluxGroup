    /// <summary>
    /// S00_Boot场景宿主：注册Mock服务→跑七步→切主场景（T001④）。
    /// 失败则停留Boot场景并打错误日志（重试页UI属T001b后；Boot校时失败策略=重试页禁卡死）。
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }
}

    public sealed class BootSequence
    {
        public const int TotalSteps = 7;
        readonly Func<bool> _switchScene;
        readonly List<string> _lines = new();
        public BootSequence(Func<bool> switchScene)
        {
            _switchScene = switchScene;
        }
        public IReadOnlyList<string> Lines => _lines;
        public bool RunAll()
        {
            bool ok = true;
            ok &= Run(1, "Logger", LoggerStep);
            ok &= Run(2, "Time", TimeStep);
            ok &= Run(3, "Config", ConfigStep);
            ok &= Run(4, "Save", SaveStep);
            ok &= Run(5, "ILocale", LocaleStep);
            ok &= Run(6, "Ads.Preload", AdsStep);
            ok &= Run(7, "SwitchMain", () => _switchScene != null && _switchScene());
            return ok;
        }
        bool Run(int n, string name, Func<bool> step)
        {
            bool result;
            try
            {
                result = step();
            }
            catch (Exception e)
            {
                Logger.Error("BOOT", $"step {n} {name} exception: {e.Message}");
                result = false;
            }
            string line = $"[BOOT] {n}/{TotalSteps} {name} {(result ? "ok" : "FAIL")}";
            _lines.Add(line);
            Logger.Info("BOOT", line);
            return result;
        }
        static bool LoggerStep() => true; // 静态门面首调即初始化；真实JS桥sink由platform层T001c注入
        static bool TimeStep()
        {
            var time = ServiceLocator.Get<ITimeService>();
            return time.ServerNowMs() > 0 && !time.TimeCheated();
        }

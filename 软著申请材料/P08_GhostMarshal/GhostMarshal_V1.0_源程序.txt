    /// S00_Boot场景宿主：注册Mock服务→跑七步→切主场景（T001④）。
    /// 失败则停留Boot场景并打错误日志（重试页UI属T001b后；Boot校时失败策略=重试页禁卡死）。

    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }
}

        void PlayTone(float freq, float seconds)
        {
            if (_audio == null) _audio = gameObject.AddComponent<AudioSource>();

        }
        static AudioClip MakeTone(float freq, float seconds)
        {
            int sr = NewbieConfig.ToneSampleRate;
            int n = Mathf.Max(1, (int)(sr * seconds));
            var data = new float[n];
            for (int i = 0; i < n; i++)
            {
                float k = 1f - (float)i / n;
                data[i] = Mathf.Sin(2f * Mathf.PI * freq * i / sr) * k;
            }
            var clip = AudioClip.Create("tone", n, 1, sr, false);
            clip.SetData(data, 0);
            return clip;
        }

        /// first tilt of every puppet in the group stays inside the 3.0s s0 window.</param>
        public PuppetMicroMotion(bool breath, bool sway, bool tilt, float phase = 0f)
        {
            Breath = breath;
            Sway = sway;
            Tilt = tilt;
            _phase = Mathf.Clamp(phase, 0f, 0.8f);

        /// <summary>Advance the motion clocks by dt (caller = the deterministic
        /// presentation seam; idempotent for dt = 0).</summary>

        {
            _wobble += dt;
            if (_wobble > TiltCycleSeconds) _wobble = 0f;

        /// <summary>Apply the enabled channels to the puppet root. Rotation is

        /// the RT gate reads (R6: localRotation != identity marks the beat).</summary>
        public void Apply(RectTransform puppet)
        {
            float t = _phase + _wobble;
            float tilt = Tilt && _wobble > TiltArmSeconds ? TiltAngle : 0f;
            float sway = Sway ? Mathf.Sin(t * ((2f * Mathf.PI) * SwayHz)) * SwayAngle : 0f;
            float angle = tilt + sway;
            puppet.localRotation = angle == 0f
                ? Quaternion.identity
                : Quaternion.Euler(0f, 0f, angle);
            if (Breath)
            {
                if (!_hasHome)
                {
                    _home = puppet.anchoredPosition;
                    _hasHome = true;
                }
                float bob = Mathf.Sin(t * ((2f * Mathf.PI) * BreathHz)) * BreathBobPixels;
                puppet.anchoredPosition = _home + new Vector2(0f, bob);
            }
        }

        /// <summary>Deterministic curtain gradient (ExploreView MakeCurtainSprite
        /// profile, trimmed: red field + fold stripes + top valance).</summary>
        static Sprite MakeCurtainSprite()
        {
            const int W = 108, H = 192;
            var tex = new Texture2D(W, H, TextureFormat.RGBA32, false);
            tex.wrapMode = TextureWrapMode.Clamp;
            var px = new Color32[W * H];
            for (int y = 0; y < H; y++)
            {
                float v = y / (float)(H - 1);
                for (int x = 0; x < W; x++)
                {
                    float u = x / (float)(W - 1);
                    float fold = 0.5f + 0.5f * Mathf.Sin(u * Mathf.PI * 9f);
                    float r = CurtainRed.r * (0.72f + 0.28f * fold);
                    float g = CurtainRed.g * (0.72f + 0.28f * fold);
                    float b = CurtainRed.b * (0.72f + 0.28f * fold);
                    float a = 0.45f + 0.25f * v;
                    float valance = Mathf.Clamp01((v - 0.855f) / 0.045f);
                    a = Mathf.Max(a, valance * 0.92f);
                    float apron = Mathf.Clamp01((0.10f - v) / 0.06f);
                    a = Mathf.Max(a, apron * 0.55f);
                    px[y * W + x] = new Color32(
                        (byte)Mathf.RoundToInt(r * 255f),
                        (byte)Mathf.RoundToInt(g * 255f),
                        (byte)Mathf.RoundToInt(b * 255f),
                        (byte)Mathf.RoundToInt(a * 255f));
                }
            }
            tex.SetPixels32(px);
            tex.Apply(false, false);
            return Sprite.Create(tex, new Rect(0, 0, W, H), new Vector2(0.5f, 0.5f), W);
        }

            var host = new GameObject("ArchiveViewHost");
            host.transform.SetParent(_root.transform, false);

                adsWatched = 0,
                adsPerUnlock = 2,
                daysRemaining = 2,
                yindeCost = 30,
            };
        }
        RectTransform FindRect(string childName)
        {
            var t = _canvas.transform.Find(childName);
            Assert.NotNull(t, "canvas child missing: " + childName);
            return (RectTransform)t;
        }
        /// <summary>One synthesized finger tap at the target's runtime
        /// screen point: touch down, one frame, touch up, then two frames
        /// so the pointer chain (module update -> EventSystem -> onClick)
        /// settles before the caller asserts.</summary>
        IEnumerator TouchTap(RectTransform target)
        {
            var screenPos = RectTransformUtility.WorldToScreenPoint(_cam, target.position);
            Assert.Greater(Screen.width, 0, "player has no screen face in this run");
            Assert.Greater(Screen.height, 0, "player has no screen face in this run");
            BeginTouch(1, screenPos);
            yield return null;
            EndTouch(1, screenPos);
            yield return null;
            yield return null;
        }

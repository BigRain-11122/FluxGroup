        /// <summary>The wave for any index: type/lanes by w%4, size by w%5
        /// (malformed negatives normalize, never throw).</summary>
        public static WaveSpec SpecFor(int waveIndex)
        {
            if (waveIndex < FirstWave) waveIndex = FirstWave;
            int m = waveIndex % 4;
            var s = new WaveSpec
            {
                waveIndex = waveIndex,
                units = 30 + (waveIndex % 5) * 2,   // 30..38 cosmetic tide size
            };
            if (m == 0 || m == 2) { s.type = WaveType.Plain; s.lanes = 1; s.guardPower = PlainGuardPower; }
            else if (m == 1) { s.type = WaveType.Wall; s.lanes = 1; s.guardPower = WallBruteGuardPower; }
            else { s.type = WaveType.Pincer; s.lanes = 3; s.guardPower = PincerGuardPower; }

        }

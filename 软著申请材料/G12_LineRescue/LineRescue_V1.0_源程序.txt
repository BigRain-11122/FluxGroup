    /// <summary>
    /// Generated-audio catalog (X027 paradigm): the AudioSetup editor
    /// tool scans Assets/_Game/Audio and serializes clip references
    /// here so runtime code resolves audio by filename stem. The asset
    /// lives at Assets/_Game/Resources/AudioCatalog.asset (Resources
    /// keeps runtime lookup build-safe with zero Addressables wiring).
    /// </summary>
    [CreateAssetMenu(fileName = "AudioCatalog", menuName = "Sy/G12 Audio Catalog")]
    public sealed class AudioCatalog : ScriptableObject
    {
        [Serializable]
        public sealed class Entry
        {
            public string key;      // filename stem, e.g. "g12_bgm_main"
            public AudioClip clip;  // direct reference keeps builds self-contained
        }
        public List<Entry> entries = new List<Entry>();
        Dictionary<string, AudioClip> _index;
        void OnEnable()
        {
            // ScriptableObject instances can be revived with a stale cache;
            // drop it so the next Get() rebuilds from entries.
            _index = null;
        }
        public AudioClip Get(string key)
        {
            if (string.IsNullOrEmpty(key)) return null;
            if (_index == null)
            {
                _index = new Dictionary<string, AudioClip>(entries.Count, StringComparer.Ordinal);
                for (int i = 0; i < entries.Count; i++)
                {
                    var e = entries[i];
                    if (e != null && !string.IsNullOrEmpty(e.key) && e.clip != null && !_index.ContainsKey(e.key))
                        _index[e.key] = e.clip;
                }
            }
            return _index.TryGetValue(key, out var c) ? c : null;
        }
    }

    /// </summary>
    public static class G12VerdictOracle
    {
        public static List<string> Problems(LineRescueSim sim)
        {
            var p = new List<string>();
            int dead = 0, eaten = 0;
            for (int i = 0; i < sim.Lines.Count; i++)
            {
                var segs = sim.Lines[i].Segs;
                for (int s = 0; s < segs.Count; s++)
                {
                    if (segs[s].Dead) dead++;
                    if (segs[s].Eaten) eaten++;
                }
            }
            if (sim.Saves != sim.CountEvents("rescue_success")) p.Add("saves!=rescue_success events");
            // AcceptRevive/AcceptRetry reset Fails to zero while the event log
            // stays append-only; each exit happened at Fails==MaxFails (Failed
            // phase stops fail production), so events = Fails +
            // (ReviveCount + FreeRetryCount) * MaxFails (v1.1 C1296: the free
            // retry exits absorb the same staircase a revive absorbs).
            if (sim.Fails + (sim.ReviveCount + sim.FreeRetryCount) * LineRescueSim.MaxFails != sim.CountEvents("rescue_fail")) p.Add("fails!=rescue_fail events");
            if (sim.DrawCount != sim.CountEvents("line_ready")) p.Add("drawCount!=line_ready events");
            if (sim.HintCount != sim.CountEvents("hint_line_shown")) p.Add("hintCount!=hint_line_shown events");
            if (sim.CountEvents("line_snap") != dead) p.Add("line_snap events!=dead segs");
            if (sim.CountEvents("eraser_eat") != eaten) p.Add("eraser_eat events!=eaten segs");
            if (sim.CurrentPhase == LineRescueSim.Phase.Failed)
            {
                if (sim.Fails < LineRescueSim.MaxFails) p.Add("failed phase with fails<MaxFails");
                if (sim.CountEvents("revive_offer") < 1) p.Add("failed phase without revive_offer");
            }
            if (sim.ReviveCount > sim.CountEvents("revive_offer")) p.Add("revives>offers");
            return p;
        }
        public static void AssertLaw(LineRescueSim sim)
        {
            List<string> probs = Problems(sim);
            Assert.IsEmpty(probs, "verdict-law oracle: " + string.Join("; ", probs.ToArray()));
        }
    }

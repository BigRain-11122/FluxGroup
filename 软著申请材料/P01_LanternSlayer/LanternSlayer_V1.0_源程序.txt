    /// <summary>
    /// S00_Boot场景宿主：注册Mock服务→跑七步→切主场景（T001④）。
    /// 失败则停留Boot场景并打错误日志（重试页UI属T001b后；Boot校时失败策略=重试页禁卡死）。
    /// </summary>
    public sealed class BootRunner : MonoBehaviour
    {
        IEnumerator Start()
        {
            BootServices.RegisterAll();
            var sequence = new BootSequence(SwitchToMain);
            if (!sequence.RunAll())
            {
                Logger.Error("BOOT", "boot sequence failed - stay on S00_Boot");
                yield break;
            }
            yield return null;
        }
        bool SwitchToMain()
        {
            StartCoroutine(LoadMainRoutine());
            return true;
        }
        IEnumerator LoadMainRoutine()
        {
            var op = SceneManager.LoadSceneAsync(SceneIds.Main, LoadSceneMode.Single);
            while (op != null && !op.isDone) yield return null;
        }
    }
}

    /// build or gate. P02 FateGate U184 pattern replicated verbatim.
    /// </summary>
    public static class UiFont
    {
        const string Fallback = "LegacyRuntime.ttf";
        static bool _warned;
        public static Font Resolve()
        {
            string name = null;
            try
            {
                if (ServiceLocator.TryGet<ILocaleService>(out var locale))
                    name = locale.GetFontSet().NormalFont;
            }
            catch (Exception) { name = null; }
            if (!string.IsNullOrEmpty(name) && name != Fallback)
            {
                var font = Resources.Load<Font>("Fonts/" + name);
                if (font != null) return font;
                if (!_warned)
                {
                    Sy.Core.Logger.Warn("UIFONT", "font asset missing, fallback LegacyRuntime: " + name);
                    _warned = true;
                }
            }
            return Resources.GetBuiltinResource<Font>(Fallback);
        }
    }
}

        public static string Get(string key, params string[] args)
        {
            bool hasArgs = args != null && args.Length > 0;
            try
            {
                if (ServiceLocator.TryGet<ILocaleService>(out var locale))
                    return locale.GetText(key, args);
            }
            catch (Exception) { /* fall through to the zero-service face */ }
            // Edit-mode/gate face: resolve the shipped table directly (the
            // AboutPanel.TryText kin; value == key counts as missing).
            string value = key;

                {

                }
            }
            return hasArgs ? string.Format(value, args) : value;

        /// <summary>Table row for a cfg point id (null = unknown id).</summary>
        public static P01Configs.AdPointRow Row(string pointId)
        {
            var table = P01Configs.AdPoint;
            if (table?.points == null) return null;
            foreach (var p in table.points)
                if (p.pointId == pointId) return p;
            return null;
        }
        /// <summary>Section-six pre-offer gate (P02 paradigm): true when the
        /// point's p01_ category still has day budget. The global 12 cap is
        /// enforced at consume; per-run caps are sim-side and owned by the host.</summary>
        public static bool PrecheckBudget(string pointId, string dayKey, AdCapPolicy policy)
        {
            var row = Row(pointId);
            if (row == null || row.type != "rewarded") return false;
            return policy.HasCategoryBudget(dayKey, AdCategories.CategoryOfPoint(AdPointId(pointId)));
        }
